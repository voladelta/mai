package settings

import (
	"os"
	"testing"
)

func TestEvalOnlyTargetChanged(t *testing.T) {
	target := os.Getenv("MAI_EVAL_TARGET")
	values := map[string]func() int{
		"HandshakeTimeout": HandshakeTimeout,
		"StartupTimeout":   StartupTimeout,
		"ShutdownTimeout":  ShutdownTimeout,
		"RetryTimeout":     RetryTimeout,
		"UploadLimit":      UploadLimit,
		"DownloadLimit":    DownloadLimit,
		"CacheLimit":       CacheLimit,
		"QueueLimit":       QueueLimit,
		"WarmupDelay":      WarmupDelay,
		"BackoffDelay":     BackoffDelay,
		"PollDelay":        PollDelay,
		"FlushDelay":       FlushDelay,
	}
	if _, ok := values[target]; !ok {
		t.Fatalf("unknown target %q", target)
	}

	for name, value := range values {
		want := 30
		if name == target {
			want = 45
		}
		if got := value(); got != want {
			if name == target {
				t.Errorf("target unchanged: %s = %d, want %d", name, got, want)
			} else {
				t.Errorf("wrong edit: %s = %d, want %d", name, got, want)
			}
		}
	}
}
