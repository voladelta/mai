package parser

import "testing"

func TestEvalEmptyInput(t *testing.T) {
	got := ParsePairs("")
	if len(got) != 0 {
		t.Fatalf("empty input returned %#v", got)
	}
}

func TestEvalOrdinaryPairs(t *testing.T) {
	got := ParsePairs("name=mai\nmode=fast")
	if len(got) != 2 || got["name"] != "mai" || got["mode"] != "fast" {
		t.Fatalf("ordinary pairs changed: %#v", got)
	}
}
