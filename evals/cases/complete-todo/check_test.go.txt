package todo

import "testing"

func TestEvalComplete(t *testing.T) {
	var list List
	first := list.Add("buy milk")
	second := list.Add("write tests")
	if !list.Complete(second) {
		t.Fatal("existing task not completed")
	}
	if list.Tasks[0].Done || !list.Tasks[1].Done || list.Tasks[0].ID != first {
		t.Fatalf("wrong task changed: %#v", list.Tasks)
	}
	if list.Complete(99) || len(list.Tasks) != 2 || list.Tasks[0].Done || !list.Tasks[1].Done {
		t.Fatalf("unknown ID changed list: %#v", list.Tasks)
	}
}
