type evalPost struct {
	ID     int    `json:"id"`
	Author string `json:"author"`
	Text   string `json:"text"`
}

func evalSocial(t *testing.T, h http.Handler, user, method, url, body string, status int) *httptest.ResponseRecorder {
	t.Helper()
	r := httptest.NewRequest(method, url, strings.NewReader(body))
	r.Header.Set("Content-Type", "application/json")
	if user != "" {
		r.Header.Set("X-User", user)
	}
	w := httptest.NewRecorder()
	h.ServeHTTP(w, r)
	if w.Code != status {
		t.Fatalf("%s %s user=%s: status=%d want=%d body=%s", method, url, user, w.Code, status, w.Body.String())
	}
	return w
}
func evalCreate(t *testing.T, h http.Handler, user, text string) evalPost {
	t.Helper()
	body, _ := json.Marshal(map[string]string{"text": text})
	w := evalSocial(t, h, user, "POST", "/posts", string(body), 201)
	var p evalPost
	if err := json.Unmarshal(w.Body.Bytes(), &p); err != nil {
		t.Fatal(err)
	}
	if p.ID <= 0 || p.Author != user || p.Text != strings.TrimSpace(text) {
		t.Fatalf("created post: %+v", p)
	}
	return p
}
func evalTimeline(t *testing.T, h http.Handler, user, query string) []evalPost {
	t.Helper()
	w := evalSocial(t, h, user, "GET", "/timeline"+query, "", 200)
	var p []evalPost
	if err := json.Unmarshal(w.Body.Bytes(), &p); err != nil {
		t.Fatal(err)
	}
	if p == nil {
		t.Fatal("null timeline")
	}
	return p
}
func TestEvalSocialTimelineAndPersistence(t *testing.T) {
	path := filepath.Join(t.TempDir(), "social.json")
	h, err := NewHandler(path)
	if err != nil {
		t.Fatal(err)
	}
	evalSocial(t, h, "", "POST", "/posts", `{"text":"x"}`, 401)
	evalSocial(t, h, "mallory", "GET", "/timeline", "", 401)
	a := evalCreate(t, h, "alice", " own ")
	b := evalCreate(t, h, "bob", "followed")
	evalCreate(t, h, "carol", "hidden")
	if p := evalTimeline(t, h, "alice", ""); len(p) != 1 || p[0].ID != a.ID {
		t.Fatalf("unfollowed leakage: %+v", p)
	}
	for i := 0; i < 2; i++ {
		evalSocial(t, h, "alice", "PUT", "/follows/bob", "", 204)
	}
	p := evalTimeline(t, h, "alice", "")
	if len(p) != 2 || p[0].ID != b.ID || p[1].ID != a.ID {
		t.Fatalf("timeline: %+v", p)
	}
	first := evalTimeline(t, h, "alice", "?limit=1")
	next := evalTimeline(t, h, "alice", fmt.Sprintf("?limit=1&before=%d", first[0].ID))
	if len(first) != 1 || len(next) != 1 || next[0].ID != a.ID {
		t.Fatal("pagination duplicate or missing post")
	}
	h, err = NewHandler(path)
	if err != nil {
		t.Fatal(err)
	}
	if len(evalTimeline(t, h, "alice", "")) != 2 {
		t.Fatal("restart lost posts or follows")
	}
	evalSocial(t, h, "alice", "DELETE", fmt.Sprintf("/posts/%d", b.ID), "", 403)
	if len(evalTimeline(t, h, "alice", "")) != 2 {
		t.Fatal("unauthorized deletion changed posts")
	}
	for i := 0; i < 2; i++ {
		evalSocial(t, h, "alice", "DELETE", "/follows/bob", "", 204)
	}
	if p := evalTimeline(t, h, "alice", ""); len(p) != 1 || p[0].ID != a.ID {
		t.Fatal("unfollow failed")
	}
	evalSocial(t, h, "alice", "DELETE", fmt.Sprintf("/posts/%d", a.ID), "", 204)
	if len(evalTimeline(t, h, "alice", "")) != 0 {
		t.Fatal("delete failed")
	}
	h, err = NewHandler(path)
	if err != nil {
		t.Fatal(err)
	}
	if p := evalCreate(t, h, "alice", "new"); p.ID <= b.ID {
		t.Fatal("ID reused after restart")
	}
	evalSocial(t, h, "alice", "PUT", "/follows/alice", "", 400)
	evalSocial(t, h, "alice", "PUT", "/follows/nobody", "", 404)
	evalSocial(t, h, "alice", "DELETE", "/posts/999999", "", 404)
	for _, query := range []string{"?limit=0", "?limit=51", "?limit=no", "?before=no"} {
		evalSocial(t, h, "alice", "GET", "/timeline"+query, "", 400)
	}
	evalSocial(t, h, "alice", "POST", "/posts", `{"text":" "}`, 400)
	evalSocial(t, h, "alice", "POST", "/posts", `{`, 400)
	evalCreate(t, h, "alice", strings.Repeat("界", 280))
	body, _ := json.Marshal(map[string]string{"text": strings.Repeat("界", 281)})
	evalSocial(t, h, "alice", "POST", "/posts", string(body), 400)
	html := evalSocial(t, h, "", "GET", "/", "", 200)
	if !strings.Contains(strings.ToLower(html.Header().Get("Content-Type")), "text/html") {
		t.Fatal("home is not HTML")
	}
	evalSocial(t, h, "alice", "PATCH", "/posts", `{}`, 405)
}

func TestEvalSocialThreadCorrection(t *testing.T) {
	path := filepath.Join(t.TempDir(), "thread.json")
	h, err := NewHandler(path)
	if err != nil {
		t.Fatal(err)
	}
	root := evalCreate(t, h, "alice", "ROOT_TEXT_MUST_BE_ERASED_927")
	base := fmt.Sprintf("/posts/%d", root.ID)
	for i := 0; i < 2; i++ {
		evalSocial(t, h, "bob", "PUT", base+"/like", "", 204)
	}
	thread := func(handler http.Handler) struct {
		Post struct {
			ID      int    `json:"id"`
			Author  string `json:"author"`
			Text    string `json:"text"`
			Likes   int    `json:"likes"`
			Deleted bool   `json:"deleted"`
		} `json:"post"`
		Replies []evalPost `json:"replies"`
	} { w := evalSocial(t, handler, "alice", "GET", base+"/thread", "", 200); var v struct {
		Post struct {
			ID      int    `json:"id"`
			Author  string `json:"author"`
			Text    string `json:"text"`
			Likes   int    `json:"likes"`
			Deleted bool   `json:"deleted"`
		} `json:"post"`
		Replies []evalPost `json:"replies"`
	}; if err := json.Unmarshal(w.Body.Bytes(), &v); err != nil {
		t.Fatal(err)
	}; return v }
	if v := thread(h); v.Post.Likes != 1 {
		t.Fatalf("duplicate like: %+v", v)
	}
	for i := 0; i < 2; i++ {
		evalSocial(t, h, "bob", "DELETE", base+"/like", "", 204)
	}
	if v := thread(h); v.Post.Likes != 0 {
		t.Fatal("unlike not idempotent")
	}
	w := evalSocial(t, h, "bob", "POST", base+"/replies", `{"text":"reply survives"}`, 201)
	var reply evalPost
	if err := json.Unmarshal(w.Body.Bytes(), &reply); err != nil || reply.ID <= root.ID {
		t.Fatalf("reply: %+v %v", reply, err)
	}
	if p := evalTimeline(t, h, "bob", ""); len(p) != 0 {
		t.Fatalf("reply leaked to root timeline: %+v", p)
	}
	evalSocial(t, h, "bob", "POST", "/posts/999999/replies", `{"text":"x"}`, 404)
	evalSocial(t, h, "bob", "DELETE", base, "", 403)
	if v := thread(h); v.Post.Deleted {
		t.Fatal("non-author deleted root")
	}
	evalSocial(t, h, "alice", "DELETE", base, "", 204)
	h, err = NewHandler(path)
	if err != nil {
		t.Fatal(err)
	}
	v := thread(h)
	if !v.Post.Deleted || v.Post.Text != "" || v.Post.ID != root.ID || v.Post.Author != "alice" || len(v.Replies) != 1 || v.Replies[0].ID != reply.ID || v.Replies[0].Text != "reply survives" {
		t.Fatalf("correction lost: %+v", v)
	}
	stored, err := os.ReadFile(path)
	if err != nil {
		t.Fatal(err)
	}
	if strings.Contains(string(stored), "ROOT_TEXT_MUST_BE_ERASED_927") {
		t.Fatal("deleted root text retained on disk")
	}
	if len(evalTimeline(t, h, "alice", "")) != 0 {
		t.Fatal("tombstone leaked into timeline")
	}
	evalSocial(t, h, "bob", "POST", base+"/replies", `{"text":"late"}`, 409)
	evalSocial(t, h, "bob", "PUT", base+"/like", "", 409)
}
